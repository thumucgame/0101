let audioContext = null; let micStream = null; let micSource = null; let processor = null; let nextPlayTime = 0; let isAiSpeaking = false; // 1. Hàm khởi tạo/kích hoạt AudioContext duy nhất cho cả Thu & Phát async function initAudioContext() { if (!audioContext || audioContext.state === "closed") { // Tạo AudioContext chung với sampleRate cố định const AudioCtx = window.AudioContext || window.webkitAudioContext; audioContext = new AudioCtx({ sampleRate: 24000 }); } if (audioContext.state === "suspended") { await audioContext.resume(); } } // 2. Phát âm thanh PCM từ AI function playPCM24k(base64) { if (!audioContext) return; isAiSpeaking = true; const bytes = base64ToBytes(base64); const samples = new Int16Array(bytes.buffer, bytes.byteOffset, Math.floor(bytes.byteLength / 2)); const buffer = audioContext.createBuffer(1, samples.length, 24000); const channel = buffer.getChannelData(0); for (let i = 0; i < samples.length; i++) { channel[i] = samples[i] / 32768; } const source = audioContext.createBufferSource(); source.buffer = buffer; source.connect(audioContext.destination); const now = audioContext.currentTime; nextPlayTime = Math.max(nextPlayTime, now); source.start(nextPlayTime); nextPlayTime += buffer.duration; } function clearPlayback() { if (audioContext) nextPlayTime = audioContext.currentTime; isAiSpeaking = false; } // 3. Khởi chạy Micro async function startMicrophone() { // Yêu cầu quyền Micro để Safari kích hoạt chế độ AudioSession thu/phát micStream = await navigator.mediaDevices.getUserMedia({ audio: { channelCount: 1, echoCancellation: true, noiseSuppression: true, autoGainControl: true } }); micSource = audioContext.createMediaStreamSource(micStream); processor = audioContext.createScriptProcessor(4096, 1, 1); processor.onaudioprocess = (event) => { if (!session || isAiSpeaking) return; const input = event.inputBuffer.getChannelData(0); const pcm = floatTo16BitPCM(input); session.sendRealtimeInput({ audio: { data: bytesToBase64(pcm), mimeType: `audio/pcm;rate=${audioContext.sampleRate}` } }); }; micSource.connect(processor); // Kết nối giả lập để giữ luồng ghi âm hoạt động trên iOS const silentGain = audioContext.createGain(); silentGain.gain.value = 0; processor.connect(silentGain); silentGain.connect(audioContext.destination); } // 4. Dừng Micro async function stopMicrophone() { try { processor?.disconnect(); } catch {} try { micSource?.disconnect(); } catch {} processor = null; micSource = null; if (micStream) { micStream.getTracks().forEach(t => t.stop()); micStream = null; } } // 5. Cập nhật hàm startConversation - kích hoạt Audio trước khi gọi API async function startConversation(targetNPC = null) { if (!geminiApiKey) { subtitle.textContent = "Enter your Gemini API key in the box above first."; geminiKeyInput.focus(); return; } try { status.textContent = "CONNECTING..."; start.disabled = true; talk.disabled = true; // BẮT BUỘC: Khởi tạo Audio Context ngay trong sự kiện Click của người dùng await initAudioContext(); await startMicrophone(); const ai = new GoogleGenAI({ apiKey: geminiApiKey }); const npcContext = targetNPC ? `You are ${targetNPC.name}, a ${targetNPC.role}, in a roleplay scenario called "${targetNPC.topic}". ${targetNPC.scenario} The learner's goal: ${targetNPC.description}` : "You are a friendly person on an English-speaking city street."; session = await ai.live.connect({ model: GEMINI_MODEL, config: { responseModalities: [Modality.AUDIO], inputAudioTranscription: {}, outputAudioTranscription: {}, thinkingConfig: { thinkingLevel: "minimal" }, systemInstruction: { parts: [{ text: `${npcContext}\nSpeak naturally and concisely.` }] } }, callbacks: { onopen() { status.textContent = "CONNECTED"; stop.disabled = false; }, onmessage(message) { const sc = message.serverContent; const parts = sc?.modelTurn?.parts || []; for (const part of parts) { const data = part?.inlineData?.data || part?.inline_data?.data; if (data) playPCM24k(data); } }, onerror(e) { console.error(e); status.textContent = "ERROR"; subtitle.textContent = "Error: " + (e?.message || "Check permissions/key."); stopMicrophone(); }, onclose() { status.textContent = "OFFLINE"; stopMicrophone(); } } }); } catch (e) { console.error(e); status.textContent = "ERROR"; subtitle.textContent = e?.message || "Cannot start Gemini Live."; await stopMicrophone(); } }
